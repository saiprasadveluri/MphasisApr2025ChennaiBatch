 3. WebAPI Controllery
[Route("api/[controller]")]
public class AuthController : ControllerBase
{
    private readonly YourDbContext _context;

    public AuthController(YourDbContext context)
    {
        _context = context;
    }

    [HttpPost("login")]
    public ActionResult<LoginResponse> Login([FromBody] LoginRequest request)
    {
        var user = _context.Users
            .FirstOrDefault(u => u.Email == request.Email && u.Password == request.Password);

        if (user == null)
        {
            return Unauthorized(new LoginResponse { IsValid = false });
        }

        return Ok(new LoginResponse
        {
            IsValid = true,
            IsAdmin = user.IsAdmin,
            UserName = user.UserName,
            UserId = user.UserId
        });
    }
}

Create Controller - AccountController
public class AccountController : Controller
{
    private readonly string apiBaseUrl = "http://your-api-base-url/";

    public ActionResult Login()
    {
        return View();
    }

    [HttpPost]
    public async Task<ActionResult> Login(LoginViewModel model)
    {
        if (!ModelState.IsValid)
            return View(model);

        using (var client = new HttpClient())
        {
            client.BaseAddress = new Uri(apiBaseUrl);
            var response = await client.PostAsJsonAsync("api/auth/login", model);

            if (response.IsSuccessStatusCode)
            {
                var result = await response.Content.ReadAsAsync<LoginResponse>();

                if (result.IsValid)
                {
                    TempData["UserName"] = result.UserName;

                    if (result.IsAdmin)
                        return RedirectToAction("Dashboard", "Admin"); // already created
                    else
                        return RedirectToAction("Dashboard", "Customer"); // create this
                }
            }

            ModelState.AddModelError("", "Invalid credentials");
            return View(model);
        }
    }
}
